// 요약: 표 기반 개인화 + 시드된 가중치 선택
import type { CanonicalInput, ProgramPlan, Block } from '@domain/types';
import vol from '@config/rules/volume.json';
import weeks from '@config/rules/weeks.json';
import freqCfg from '@config/rules/frequency.json';
import varW from '@config/rules/variationWeights.json';
import schemes from '@config/schemes/presets.json';

export function planFromTables(input: CanonicalInput): ProgramPlan {
  const seed = hash(input.meta.raw_hash);        // 사용자 고유 시드
  const rng = mulberry32(seed);

  // 1) 볼륨/빈도/주차% 결정
  const setsW = chooseSets(vol, input);          // {SQ:16,BP:22,DL:10} 같은 형태
  const freq  = chooseFrequency(freqCfg, input); // {SQ:3,BP:5,DL:2}
  const weekPct = pickWeeks(weeks, input);       // [0.68,0.72,0.76,0.62] 등

  // 2) 주차×요일 스켈레톤(Heavy/Mod/Light)
  const frame = splitIntoDays(setsW, freq, weekPct.length); // [{week,day,dayType,targets:{SQ:5,BP:4…}}]

  // 3) 각 셀에 스킴 + 변형/액세서리 채우기
  const blocks: Block[] = [];
  for (const d of frame) {
    const lifts: ('SQ'|'BP'|'DL')[] = ['SQ','BP','DL'];
    for (const L of lifts) {
      const sets = d.targets[L] || 0; if (!sets) continue;

      // 3-1) 변형 선택(가중치 + 시드)
      const ex = weightedPick(varW[L], input, rng); // 예: 'pause_squat' or 'close_grip_bench'

      // 3-2) 스킴 선택(Heavy/Mod/Light 매핑)
      const scheme = schemeByDayType(L, d.dayType, schemes);

      // 3-3) 주차 강도 반영해 Program row들 생성
      const rows = compileScheme({
        week: d.week, day: d.day, lift: L, oneRM: get1RM(input,L),
      }, materializeScheme(scheme, sets, weekPct[d.week-1]));

      blocks.push(...rows);
    }

    // 3-4) 액세서리 채우기(회전, 8–12 reps)
    blocks.push(...buildAccessories(input, d, rng));
  }

  // 4) 안전 가드(캡/클램프)
  const safe = guardRails(blocks); // 세션 최대세트≤16, ≥85% 세션/주 ≤2, RPEcap 등

  return { meta:{ mesoWeeks: weekPct.length, engineVersion:'rules-v3', createdAt: new Date().toISOString() }, blocks: safe };
}

// --- 유틸(핵심) ---
function hash(s:string){ let h=1779033703; for (let i=0;i<s.length;i++) { h = (h ^ s.charCodeAt(i)) * 3432918353; h = (h<<13) | (h>>>19); } return h>>>0; }
function mulberry32(a:number){ return function(){ a|=0; a = a + 0x6D2B79F5 |0; let t = Math.imul(a ^ a>>>15, 1 | a); t = t + Math.imul(t ^ t>>>7, 61 | t) ^ t; return ((t ^ t>>>14) >>> 0) / 4294967296; }; }
function weightedPick(table:any, input:CanonicalInput, rng:()=>number){
  // table 예: varW.SQ.depth = { pause_squat:3, box_squat:2, front_squat:1 } + 생체치수/문제 반영
  const weights = scoreVariants(table, input);          // {variant:score}
  const total = Object.values(weights).reduce((a,b)=>a+b,0);
  let r = rng()*total; for (const [k,w] of Object.entries(weights)){ if ((r-=Number(w))<=0) return k; } return Object.keys(weights)[0];
}
